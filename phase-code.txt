function readJsonBody<T>(req: IncomingMessage, maxBytes = 1_000_000): Promise<T> {
  return new Promise((resolve, reject) => {
    // ۱. چک کردن Content-Type
    if (!req.headers['content-type']?.includes('application/json')) {
      reject(new Error('فقط JSON پذیرفته می‌شود'));
      return;
    }

    let size = 0;
    const chunks: Buffer[] = [];

    req.on('data', (chunk: Buffer) => {
      size += chunk.length;
      // ۲. محدود کردن حجم (ضد DoS)
      if (size > maxBytes) {
        reject(new Error('حجم درخواست بیش از حد مجاز'));
        req.destroy();   // اتصال را قطع کن
        return;
      }
      chunks.push(chunk);
    });

    req.on('end', () => {
      try {
        resolve(JSON.parse(Buffer.concat(chunks).toString('utf-8')));
      } catch {
        reject(new Error('JSON نامعتبر'));
      }
    });

    req.on('error', reject);
  });
}

import { createServer, IncomingMessage, ServerResponse } from 'node:http';

type Handler = (req: IncomingMessage, res: ServerResponse) => Promise<void> | void;

const routes = new Map<string, Map<string, Handler>>();

// ثبت مسیر
function on(method: string, path: string, handler: Handler) {
  if (!routes.has(method)) routes.set(method, new Map());
  routes.get(method)!.set(path, handler);
}

// تطبیق و اجرا
const server = createServer(async (req, res) => {
  const url = new URL(req.url!, `http://${req.headers.host}`);
  const handler = routes.get(req.method!)?.get(url.pathname);

  if (!handler) {
    res.writeHead(404, { 'Content-Type': 'application/json' });
    res.end(JSON.stringify({ error: 'یافت نشد' }));
    return;
  }

  try {
    await handler(req, res);
  } catch (err) {
    res.writeHead(500, { 'Content-Type': 'application/json' });
    res.end(JSON.stringify({ error: 'خطای داخلی سرور' }));
  }
});
